import type { AILetterResponse, ConversationResponse } from "@pharmacist-tree-hollow/shared";

const FALLBACK_CARE_TITLE = "今晚這張紙先放著";
const MAX_CARE_TITLE_LEN = 14;

function truncateTitle(title: string): string {
  const chars = [...title];
  if (chars.length <= MAX_CARE_TITLE_LEN) return title;
  return chars.slice(0, MAX_CARE_TITLE_LEN).join("");
}

export function staticToAILetter(input: ConversationResponse): AILetterResponse {
  const isCrisis = input.riskLevel === "crisis";

  const careTitleRaw = input.careTitle && input.careTitle.length > 0
    ? input.careTitle
    : FALLBACK_CARE_TITLE;

  const hold = input.empathy && input.empathy.length > 0
    ? input.empathy
    : input.message.join(" ");

  return {
    mode: isCrisis ? "crisis" : "hold_and_praise",
    careTitle: truncateTitle(careTitleRaw),
    hold,
    praise: isCrisis ? "" : input.praise,
    praiseNotes: isCrisis ? [] : (input.praiseNotes ?? [])
  };
}
```

import type { AILetterResponse } from "@pharmacist-tree-hollow/shared";
import type { MoodTag } from "@pharmacist-tree-hollow/shared";
import { buildResponse } from "./respond";
import { staticToAILetter } from "./letter-adapter";

const VALID_MODES = new Set(["hold_and_praise", "praise_only", "hold_only", "crisis"]);

export type LetterSource = "ai" | "fallback";

export interface LetterResult {
  letter: AILetterResponse;
  source: LetterSource;
}

function looksLikeAILetter(value: unknown): value is AILetterResponse {
  if (typeof value !== "object" || value === null) return false;
  const v = value as Record<string, unknown>;
  return (
    typeof v.mode === "string" &&
    VALID_MODES.has(v.mode) &&
    typeof v.careTitle === "string" &&
    typeof v.hold === "string" &&
    typeof v.praise === "string" &&
    Array.isArray(v.praiseNotes)
  );
}

function fallback(input: string, mood: MoodTag): LetterResult {
  const staticResponse = buildResponse(input, mood);
  return {
    letter: staticToAILetter(staticResponse),
    source: "fallback"
  };
}

export async function requestAILetter(input: string, mood: MoodTag): Promise<LetterResult> {
  try {
    const response = await fetch("/api/respond", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ input, mood })
    });

    if (!response.ok) return fallback(input, mood);

    const data = await response.json();
    if (!looksLikeAILetter(data)) return fallback(input, mood);

    return { letter: data, source: "ai" };
  } catch {
    return fallback(input, mood);
  }
}
```

async function submit() {
  const trimmed = input.trim();
  timersRef.current.forEach((timer) => window.clearTimeout(timer));
  timersRef.current = [];

  setIsDepositing(true);
  setIsThinking(false);
  setIsResponding(false);
  setResponse(null);

  await new Promise((r) => setTimeout(r, 600));
  setIsDepositing(false);
  setIsThinking(true);

  const userText = trimmed || `今天覺得${mood}`;
  const safety = classifySafety(userText);

  let letter: AILetterResponse;

  if (safety.riskLevel === "crisis") {
    const staticCrisis = buildResponse(userText, mood);
    letter = staticToAILetter({ ...staticCrisis, riskLevel: "crisis" });
  } else {
    const safeText = redactSensitiveText(userText);
    const result = await requestAILetter(safeText, mood);
    letter = result.letter;
  }

  setIsThinking(false);
  setResponse(letter);
  setIsResponding(true);

  const respondingTimer = window.setTimeout(() => setIsResponding(false), 600);
  timersRef.current.push(respondingTimer);
}
